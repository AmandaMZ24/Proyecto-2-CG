# escena_materiales.txt

# Escena de prueba del modelo de iluminacion completo:
#   - iluminacion ambiental (ambiente)
#   - reflexion difusa de Lambert
#   - reflexion especular de Phong   sum((R.V)^kn * Ks * Ip * Fatt)
#   - sombras proyectadas sobre el suelo y entre objetos
#   - materiales distintos por objeto
#
# Sintaxis de material (las dos formas soportadas):
#   esfera     ... <r> <g> <b> <ka> <kd> <ks> <kn>
#   poligono   ... <r> <g> <b> <ka> <kd> <ks> <kn>
#   material   <ka> <kd> <ks> <kn>      (aplica al objeto anterior)
# ============================================================

resolucion 1300 760

ojo 0 0 -5
ventana -3.2 -1.87 3.2 1.87

fondo 0.05 0.05 0.15

# Intensidad de la iluminacion ambiental
ambiente 0.5

# Luz principal: arriba y a la izquierda. Es la que proyecta
# las sombras de los objetos sobre el suelo.
luz -2.5 4.0 1.5 1.0

# Luz frontal derecha: aporta la mayor parte del brillo especular.
luz 3.0 1.8 -2.5 0.6

# Luz trasera: ilumina la parte superior de las esferas y el suelo.
luz 0.0 1.5 6.5 0.4

# ------------------------------------------------------------
# SUELO (poligono convexo horizontal, normal +Y)
# ------------------------------------------------------------
poligono 4 -200 -2.75 300  200 -2.75 300  200 -2.75 -4  -200 -2.75 -4  0.55 0.57 0.62  0.25 0.70 0.35 16

# ------------------------------------------------------------
# MURO CUADRADO (poligono convexo vertical)
# ------------------------------------------------------------
poligono 4 -4.5 -2.75 5.5  -1.5 -2.75 5.5  -1.5 0.5 5.5  -4.5 0.5 5.5  0.90 0.25 0.20
material 0.40 0.90 0.05 8

# ------------------------------------------------------------
# LETRA L (poligono CONCAVO de 6 vertices)
# ------------------------------------------------------------
poligono 6 1.0 -2.75 6.0  3.4 -2.75 6.0  3.4 -1.6 6.0  2.2 -1.6 6.0  2.2 0.6 6.0  1.0 0.6 6.0  0.10 0.78 0.32  0.30 0.85 0.45 48

# ------------------------------------------------------------
# ESFERAS (12) descansando sobre el suelo
# ------------------------------------------------------------
# Roja: plastico brillante (material al final de la linea)
esfera -3.2 -1.45 3.0 1.3  0.98 0.20 0.25   0.20 0.70 0.90 128

# Azul: mate, casi sin especular
esfera 0.0 -1.65 4.5 1.1  0.12 0.48 0.97   0.35 0.85 0.02 12

# Morada: metalica, brillo muy concentrado
esfera -1.6 -1.9 6.5 0.85  0.55 0.39 0.96  0.15 0.50 1.00 256

# Turquesa: ceramica
esfera 2.6 -1.8 3.6 0.95  0.11 0.77 0.77   0.20 0.75 0.85 96

# Amarilla: difusa pura, sin especular
esfera -5.0 -2.2 4.6 0.55  1.00 0.89 0.12   0.40 0.90 0.00 8

# Verde: material por defecto (sin coeficientes)
esfera 4.6 -2.15 5.2 0.6  0.27 0.91 0.22

# Naranja: plastico
esfera 1.4 -2.3 7.6 0.45  1.00 0.54 0.19   0.25 0.80 0.55 64

# Azul profundo: acetato
esfera -0.9 -2.05 1.6 0.7  0.26 0.42 0.86   0.20 0.70 0.60 40

# Violeta: satinado
esfera 3.9 -2.35 7.0 0.4  0.73 0.25 0.84   0.20 0.75 0.70 80

# Coral
esfera -2.4 -2.25 8.4 0.5  0.97 0.39 0.32  0.30 0.80 0.35 24

# Blanca: espejo suave
esfera 0.9 -1.25 9.5 1.5  0.98 0.96 0.89  0.35 0.70 0.95 200

# Fucsia
esfera 5.2 -2.25 8.8 0.5  0.94 0.13 0.57   0.25 0.80 0.45 48

fin
